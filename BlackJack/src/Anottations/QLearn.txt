static void trainAgent(){
  Printer.printLn(">>>> TRAINING");
  int nextMilestone = 0;

  for (int i = 1; i <= TRAIN_STEPS; i++) {
    // PROGRESS LOG
    int progress = (i * 100) / TRAIN_STEPS;
    if (progress >= nextMilestone) {
      Printer.printLn(nextMilestone + "%");
      nextMilestone += 10;
    }

    // RUN GAME
    gameHandler.startGame();
    GameState prevState = null;
    int prevAction = -1;
    while (!gameHandler.isGameEnded()) {
      GameState currentState = gameHandler.getGameState();
      int action = agent.chooseAction(currentState);
      gameHandler.takeAction(action == agent.HIT ? PlayerAction.HIT : PlayerAction.STAND);
      if (prevState != null && !gameHandler.isGameEnded()) {
        GameState newState = gameHandler.getGameState();
        agent.updateStep(prevState, prevAction, newState);
      }
      prevState = currentState;
      prevAction = action;
    }

    // EVALUATE REWARD
    int reward = 0;
    switch (gameHandler.getGameResult()){
      case PLAYER_WIN -> reward = 1;
      case DEALER_WIN -> reward = -1;
    }
    if (prevState != null) agent.updateTerminal(prevState, prevAction, reward);
  }
}
