# QLEARN TRAINING =============================================
static void trainAgent(){
  Printer.printLn(">>>> TRAINING");
  int nextMilestone = 0;

  for (int i = 1; i <= TRAIN_STEPS; i++) {
    // PROGRESS LOG
    int progress = (i * 100) / TRAIN_STEPS;
    if (progress >= nextMilestone) {
      Printer.printLn(nextMilestone + "%");
      nextMilestone += 10;
    }

    // RUN GAME
    gameHandler.startGame();
    GameState prevState = gameHandler.getGameState();
    int prevAction = agent.chooseAction(gameHandler.getGameState());
    while (!gameHandler.isGameEnded()) {
      gameHandler.takeAction(prevAction == agent.HIT ? PlayerAction.HIT : PlayerAction.STAND);
      if (gameHandler.isGameEnded()) break;

      GameState nextState = gameHandler.getGameState();
      int nextAction = agent.chooseAction(nextState);
      agent.updateStep(prevState, prevAction, nextState, nextAction, 0);

      prevState = nextState;
      prevAction = nextAction;
    }

    // EVALUATE REWARD
    int reward = 0;
    switch (gameHandler.getGameResult()){
      case PLAYER_WIN -> reward = 1;
      case DEALER_WIN -> reward = -1;
    }
    if (prevState != null) agent.updateTerminal(prevState, prevAction, reward);
  }
}
